Build:
flutter clean
flutter pub get
flutter build apk --release

For iOS:
flutter clean
rm -rf ios/Flutter/ephemeral ios/Pods ios/Podfile.lock .dart_tool build
flutter pub get
cd ios && pod install --repo-update && cd ..
open ios/Runner.xcworkspace
